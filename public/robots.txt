# work-on-the-moon — robots policy
#
# This server hosts a single-user passkey-protected web UI plus a public
# marketing landing page. Only the landing page and its public assets are
# indexable. Authenticated app routes are blocked from crawling.

User-agent: *
Allow: /landing
Allow: /llms.txt
Allow: /sitemap.xml
Allow: /static/landing.css
Allow: /static/apple-touch-icon.png
Disallow: /chat/
Disallow: /chat-live/
Disallow: /api/
Disallow: /setup
Disallow: /devices
Disallow: /login
